pub fn join(parts: vec<txt>, separator: txt) -> txt {
  let result: txt = "";
  let first: bool = true;
  for part in parts {
    match first {
      true => { result = part; first = false; };
      _ => { result = "@(result)@(separator)@(part)"; };
    };
  };
  return result;
};

fn all_command(kiru: Project, docs: Project, flag: txt) {
  match flag {
    "build" => {
      std::log("Building site");
      async build_docs(docs);
      std::log("Building backend");
      async build_kiru(kiru);
      wait;
    };
    _ => { std::eprint(usage_all); };
  };
};

pub fn eprint(message: txt) {

  panic;
};
