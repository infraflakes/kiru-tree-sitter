let padded: txt = "  hi  ";
let trailing: txt = "  hi\n";
let empty: txt = "";
let literal_at: txt = "user@host";

# A string may span lines.
let usage: txt = "usage: kiru <flag>

flags:
  check      run the checks
  build      build the binary";
