fn prepend<T>(self: vec<T>, item: T) -> vec<T> {
  let out: vec<T> = [];
  out[0] = item;
  let index: int = 0;
  for element in self {
    index = index + 1;
    out[index] = element;
  };
  return out;
};

fn reverse<T>(self: vec<T>) -> vec<T> {
  let result: vec<T> = [];
  let at: int = 0;
  let index: int = self.len();
  for {
    match index > 0 { false => { break; }; true => {}; };
    index = index - 1;
    result[at] = self[index];
    at = at + 1;
  };
  return result;
};

fn count_where<T>(items: vec<T>, keep: fn(T) -> bool) -> int {
  let total: int = 0;
  for item in items {
    match keep(item) {
      true => { total = total + 1; };
      _ => {};
    };
  };
  return total;
};
