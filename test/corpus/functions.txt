fn flag_of(context: std::cli::Context) -> txt {
  let words: vec<txt> = context.args();
  let flag: txt = match words.len() { 0 => ""; _ => words[0]; };
  return flag;
};

fn probe(service: Service, spool: txt) {

};

fn container_cmd() -> txt {
  match std::command("command -v docker").execute() {

  };

fn count_where<T>(items: vec<T>, keep: fn(T) -> bool) -> int {
  let total: int = 0;
  for item in items {
    match keep(item) {
      true => { total = total + 1; };
      _ => {};
    };
  };
  return total;
};

fn map<T, U>(self: vec<T>, apply: fn(T) -> U) -> vec<U> {
  let out: vec<U> = [];
  let index: int = 0;
  for item in self {
    out[index] = apply(item);
    index = index + 1;
  };
  return out;
};

fn healthy(health: Health) -> bool {
  return match health { Health.down(_) => false; _ => true; };
};
