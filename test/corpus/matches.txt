fn health_of(name: txt, spool: txt) -> Health {
  let raw: txt = std::fs::read(std::path::join(spool, name));
  let parts: vec<txt> = raw.split("|");
  let health: Health = match parts[0] {
    "ok" => Health.ok(parts[1]);
    "slow" => Health.slow(parts[1]);
    _ => Health.down(parts[1]);
  };
  return health;
};

fn unwrap_or<T>(self: Option<T>, fallback: T) -> T {
  let inner: T = match self {
    Option.some(value) => value;
    Option.none => fallback;
  };
  return inner;
};

fn line(self: Health) -> txt {
  let text: txt = match self {
    Health.ok(why) => "ok (@(why))";
    Health.slow(why) => "slow (@(why))";
    Health.down(why) => "down: @(why)";
  };
  return text;
};

fn signal_number(name: txt) -> int {
  let number: int = 0;
  match name {
    "TERM" => { number = 15; };
    "KILL" => { number = 9; };
    _ => { std::eprint("unknown signal `@(name)`"); };
  };
  return number;
};

fn healthy(health: Health) -> bool {
  let good: bool = match health {
    Health.down(_) => false;
    _ => true;
  };
  return good;
};

fn probe(service: Service, spool: txt) {
  let verdict: txt = match run.code {
    0 => match run.out.contains("slow") {
      true => "slow";
      _ => "ok";
    };
    _ => "down";
  };
  std::fs::write(spool, verdict);
};

pub fn emit(color: txt, prefix: txt, file_descriptor: int, message: txt) {
  std::io::write(file_descriptor, match std::io::is_terminal(file_descriptor) {
    true => "\e[@(color)m@(prefix)\e[0m @(message)\n";
    _ => "@(prefix) @(message)\n";
  });
};

fn to_int(negative: bool) -> int {
  return match negative { true => 1; false => 0; };
};
